cmake_minimum_required(VERSION 3.21)
project(ohos_codeblocks VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Xml)
add_library(ide_core STATIC src/project.cpp src/project.h src/runner.cpp src/runner.h)
target_include_directories(ide_core PUBLIC src)
target_link_libraries(ide_core PUBLIC Qt6::Core Qt6::Xml)
add_executable(ohos_codeblocks WIN32 src/main.cpp src/mainwindow.cpp src/mainwindow.h
    src/codeeditor.cpp src/codeeditor.h resources/resources.qrc)
target_link_libraries(ohos_codeblocks PRIVATE ide_core Qt6::Widgets)
install(TARGETS ohos_codeblocks RUNTIME DESTINATION bin)
install(DIRECTORY examples/ DESTINATION share/ohos_codeblocks/examples)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
    add_executable(process_fixture tests/process_fixture.cpp)
    target_link_libraries(process_fixture PRIVATE Qt6::Core)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE ide_core Qt6::Test)
    add_dependencies(core_tests process_fixture)
    add_test(NAME core_tests COMMAND core_tests)
    add_test(NAME gui_smoke COMMAND ohos_codeblocks --smoke-test)
    set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
